from __future__ import annotations

import ipaddress
import mimetypes
import os
import socket
import ssl
import threading
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from typing import Any, Callable, Mapping
from urllib.parse import quote, urlsplit


TEXTUAL_CONTENT_TYPES = {
    "application/javascript",
    "application/json",
    "application/x-csh",
    "application/x-sh",
    "application/xml",
}
TEXTUAL_SUFFIXES = {
    ".conf",
    ".css",
    ".csv",
    ".env",
    ".htm",
    ".html",
    ".ini",
    ".js",
    ".json",
    ".log",
    ".md",
    ".py",
    ".sh",
    ".sql",
    ".txt",
    ".xml",
    ".yaml",
    ".yml",
}


@dataclass(frozen=True)
class FileResponse:
    path: Path
    binary: bool | None = None
    content_type: str | None = None
    download_name: str | None = None


@dataclass(frozen=True)
class ResolvedResponse:
    body: bytes
    content_type: str
    download_name: str | None = None


def readFile(
    path: str | os.PathLike[str],
    binary: bool | None = None,
    content_type: str | None = None,
    download_name: str | None = None,
) -> FileResponse:
    return FileResponse(
        path=Path(path),
        binary=binary,
        content_type=content_type,
        download_name=download_name,
    )


def read_file(
    path: str | os.PathLike[str],
    binary: bool | None = None,
    content_type: str | None = None,
    download_name: str | None = None,
) -> FileResponse:
    return readFile(
        path=path,
        binary=binary,
        content_type=content_type,
        download_name=download_name,
    )


class Server:
    def __init__(
        self,
        schema: str,
        port: int,
        response: Any,
        code: int = 200,
        domain: str | None = None,
        path: str = "/",
        header: Mapping[str, Any] | list[tuple[str, Any]] | None = None,
        CRLF: bool = False,
        host: str = "0.0.0.0",
        cert_dir: str | os.PathLike[str] | None = None,
    ) -> None:
        self.schema = schema.lower()
        if self.schema not in {"http", "https"}:
            raise ValueError("schema must be 'http' or 'https'")

        if response is None:
            raise ValueError("response is required")

        self.port = int(port)
        self.response = response
        self.code = self._normalize_status_code(code)
        self.domain = domain
        self.path = self._normalize_prefix(path)
        self.headers = self._normalize_headers(header)
        self.CRLF = bool(CRLF)
        self.host = host
        self.base_dir = Path(__file__).resolve().parent
        self.cert_dir = (
            Path(cert_dir)
            if cert_dir is not None
            else self.base_dir / "certs" / self._safe_name(f"{self.domain or self.host}_{self.port}")
        )
        self._httpd: ThreadingHTTPServer | None = None
        self._thread: threading.Thread | None = None

    @property
    def url(self) -> str:
        host = self.domain or self.host
        return f"{self.schema}://{host}:{self.port}{self.path}"

    def start(self, background: bool = True) -> "Server":
        if self._httpd is not None:
            return self

        handler_cls = self._build_handler()
        httpd = ThreadingHTTPServer((self.host, self.port), handler_cls)

        if self.schema == "https":
            context = self._create_ssl_context()
            httpd.socket = context.wrap_socket(httpd.socket, server_side=True)

        self._httpd = httpd

        if background:
            self._thread = threading.Thread(
                target=httpd.serve_forever,
                name=f"{self.schema}:{self.port}",
                daemon=True,
            )
            self._thread.start()
            return self

        self.serve_forever()
        return self

    def serve_forever(self) -> None:
        if self._httpd is None:
            self.start(background=False)
            return

        self._httpd.serve_forever()

    def stop(self) -> None:
        if self._httpd is None:
            return

        self._httpd.shutdown()
        self._httpd.server_close()
        self._httpd = None
        self._thread = None

    def join(self, timeout: float | None = None) -> None:
        if self._thread is not None:
            self._thread.join(timeout=timeout)

    @classmethod
    def start_all(cls, *servers: "Server") -> list["Server"]:
        started: list[Server] = []
        for server in servers:
            started.append(server.start(background=True))
        return started

    @classmethod
    def join_all(cls, *servers: "Server") -> None:
        for server in servers:
            server.join()

    def _build_handler(self) -> type[BaseHTTPRequestHandler]:
        server_config = self

        class ConfigurableHandler(BaseHTTPRequestHandler):
            server_version = "ConfigurableServer/1.0"

            def handle_one_request(self) -> None:
                self.raw_requestline = self.rfile.readline(65537)
                if len(self.raw_requestline) > 65536:
                    self.requestline = ""
                    self.request_version = ""
                    self.command = ""
                    self.send_error(414)
                    return

                if not self.raw_requestline:
                    self.close_connection = True
                    return

                if not self.parse_request():
                    return

                self._dispatch()

            def _dispatch(self) -> None:
                server_config._log_request(self)
                request_path = urlsplit(self.path).path or "/"
                if not server_config._matches(request_path):
                    self.close_connection = True
                    self.send_error(404, "Path prefix does not match")
                    return

                resolved = server_config._resolve_response(self)
                response_headers = server_config._build_response_headers(resolved)

                self.close_connection = True
                self.send_response(server_config.code)
                for key, value in response_headers.items():
                    self.send_header(key, value)
                self.end_headers()

                if self.command != "HEAD":
                    self.wfile.write(resolved.body)

            def log_message(self, format: str, *args: Any) -> None:
                return

        return ConfigurableHandler

    def _matches(self, request_path: str) -> bool:
        if self.path == "/":
            return True
        return request_path.startswith(self.path)

    def _log_request(self, handler: BaseHTTPRequestHandler) -> None:
        client_ip, client_port = handler.client_address[:2]
        print(f"[{self._build_request_url(handler)}] from {client_ip}:{client_port}")

    def _build_request_url(self, handler: BaseHTTPRequestHandler) -> str:
        host = handler.headers.get("Host")
        if not host:
            host = self.domain or f"{self.host}:{self.port}"
        return f"{self.schema}://{host}{handler.path}"

    def _resolve_response(self, handler: BaseHTTPRequestHandler) -> ResolvedResponse:
        response = self.response
        if callable(response):
            response = self._call_response_factory(response, handler)

        if isinstance(response, FileResponse):
            return self._resolve_file_response(response)

        if isinstance(response, str):
            return ResolvedResponse(
                body=self._apply_crlf(response.encode("utf-8"), is_text=True),
                content_type="text/plain; charset=utf-8",
            )

        if isinstance(response, (bytes, bytearray, memoryview)):
            return ResolvedResponse(
                body=self._apply_crlf(bytes(response), is_text=False),
                content_type="application/octet-stream",
            )

        raise TypeError(
            "response must be str, bytes-like, FileResponse, or a callable returning one of them"
        )

    def _call_response_factory(
        self,
        factory: Callable[..., Any],
        handler: BaseHTTPRequestHandler,
    ) -> Any:
        try:
            return factory(handler)
        except TypeError:
            return factory()

    def _resolve_file_response(self, file_response: FileResponse) -> ResolvedResponse:
        file_path = self._resolve_path(file_response.path)
        body = file_path.read_bytes()

        content_type = file_response.content_type
        if content_type is None:
            guessed_content_type, _ = mimetypes.guess_type(file_path.name)
            if guessed_content_type is None:
                content_type = (
                    "application/octet-stream"
                    if file_response.binary
                    else "text/plain; charset=utf-8"
                )
            elif self._is_textual_content_type(guessed_content_type):
                content_type = f"{guessed_content_type}; charset=utf-8"
            else:
                content_type = guessed_content_type

        is_text = self._is_text_file(file_response, content_type, file_path)
        return ResolvedResponse(
            body=self._apply_crlf(body, is_text=is_text),
            content_type=content_type,
            download_name=file_response.download_name or file_path.name,
        )

    def _build_response_headers(self, resolved: ResolvedResponse) -> dict[str, str]:
        headers = {
            "Content-Type": resolved.content_type,
            "Content-Length": str(len(resolved.body)),
            "Connection": "close",
        }

        if resolved.download_name:
            headers["Content-Disposition"] = (
                f"attachment; filename=\"{resolved.download_name}\"; "
                f"filename*=UTF-8''{quote(resolved.download_name)}"
            )

        for key, value in self.headers.items():
            headers[key] = str(value)

        return headers

    def _apply_crlf(self, body: bytes, is_text: bool) -> bytes:
        if not self.CRLF or not is_text:
            return body
        return body.replace(b"\r\n", b"\n").replace(b"\r", b"\n")

    def _is_text_file(
        self,
        file_response: FileResponse,
        content_type: str,
        file_path: Path,
    ) -> bool:
        if file_response.binary is True:
            return False
        if file_response.binary is False:
            return True

        content_type_without_charset = content_type.split(";", 1)[0].strip().lower()
        return self._is_textual_content_type(content_type_without_charset) or (
            file_path.suffix.lower() in TEXTUAL_SUFFIXES
        )

    def _is_textual_content_type(self, content_type: str) -> bool:
        return content_type.startswith("text/") or content_type in TEXTUAL_CONTENT_TYPES

    def _resolve_path(self, value: Path) -> Path:
        if value.is_absolute():
            return value
        return (self.base_dir / value).resolve()

    def _normalize_prefix(self, prefix: str) -> str:
        prefix = prefix or "/"
        if not prefix.startswith("/"):
            prefix = "/" + prefix
        return prefix

    def _normalize_headers(
        self,
        header: Mapping[str, Any] | list[tuple[str, Any]] | None,
    ) -> dict[str, Any]:
        if header is None:
            return {}
        if isinstance(header, Mapping):
            return dict(header)
        return {key: value for key, value in header}

    def _normalize_status_code(self, code: int) -> int:
        code = int(code)
        if not 100 <= code <= 999:
            raise ValueError("code must be between 100 and 999")
        return code

    def _create_ssl_context(self) -> ssl.SSLContext:
        cert_file, key_file = self._ensure_certificate_files()
        context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        context.load_cert_chain(certfile=str(cert_file), keyfile=str(key_file))
        return context

    def _ensure_certificate_files(self) -> tuple[Path, Path]:
        self._require_cryptography()

        cert_file = self.cert_dir / "cert.pem"
        key_file = self.cert_dir / "key.pem"
        if cert_file.exists() and key_file.exists():
            return cert_file, key_file

        self.cert_dir.mkdir(parents=True, exist_ok=True)
        self._generate_self_signed_cert(cert_file, key_file)
        return cert_file, key_file

    def _generate_self_signed_cert(self, cert_file: Path, key_file: Path) -> None:
        x509, hashes, serialization, rsa, NameOID = self._require_cryptography()

        key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        common_name = self.domain or "localhost"
        subject = issuer = x509.Name(
            [
                x509.NameAttribute(NameOID.COUNTRY_NAME, "CN"),
                x509.NameAttribute(NameOID.ORGANIZATION_NAME, "Configurable Local Server"),
                x509.NameAttribute(NameOID.COMMON_NAME, common_name),
            ]
        )

        now = datetime.now(timezone.utc)
        certificate = (
            x509.CertificateBuilder()
            .subject_name(subject)
            .issuer_name(issuer)
            .public_key(key.public_key())
            .serial_number(x509.random_serial_number())
            .not_valid_before(now - timedelta(minutes=1))
            .not_valid_after(now + timedelta(days=3650))
            .add_extension(self._build_san_entries(x509), critical=False)
            .add_extension(x509.BasicConstraints(ca=True, path_length=None), critical=True)
            .sign(private_key=key, algorithm=hashes.SHA256())
        )

        key_file.write_bytes(
            key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.TraditionalOpenSSL,
                encryption_algorithm=serialization.NoEncryption(),
            )
        )
        cert_file.write_bytes(certificate.public_bytes(serialization.Encoding.PEM))

    def _build_san_entries(self, x509_module: Any) -> Any:
        names: list[Any] = []
        raw_hosts = {
            "localhost",
            "127.0.0.1",
            self.host,
            self.domain or "",
            socket.gethostname(),
            socket.getfqdn(),
        }

        for item in sorted(host for host in raw_hosts if host):
            try:
                names.append(x509_module.IPAddress(ipaddress.ip_address(item)))
            except ValueError:
                names.append(x509_module.DNSName(item))

        return x509_module.SubjectAlternativeName(names)

    def _safe_name(self, value: str) -> str:
        return "".join(ch if ch.isalnum() or ch in {"-", "_", "."} else "_" for ch in value)

    def _require_cryptography(self) -> tuple[Any, Any, Any, Any, Any]:
        try:
            from cryptography import x509
            from cryptography.hazmat.primitives import hashes, serialization
            from cryptography.hazmat.primitives.asymmetric import rsa
            from cryptography.x509.oid import NameOID
        except ImportError as exc:
            raise RuntimeError(
                "HTTPS mode requires cryptography: pip install cryptography"
            ) from exc

        return x509, hashes, serialization, rsa, NameOID
